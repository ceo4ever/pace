import json, hashlib, os, shutil
F = "/private/tmp/pace-P012-work/T03-fixture-author/fixture"
def sha(s): return hashlib.sha256(s.encode()).hexdigest()
def dump(rows): return "".join(json.dumps(r, ensure_ascii=False, separators=(",", ":")) + "\n" for r in rows)
def load(p): return [json.loads(l) for l in open(p, encoding="utf-8") if l.strip()]
# 1) P602 role start rows
p = F + "/projects/P602/log.jsonl"
rows = load(p)
for r in rows:
    if r["type"] == "start" and "role" in r:
        n = r["role"]["name"]
        r["by"] = "cli"
        r["role"] = dict(name=n, source="option", definition=".pace/team/agents/" + n + ".md",
                         hash=sha("synthetic-definition:" + n), method="prompt", model="synthetic-model",
                         effort="medium", settingsStatus="planned",
                         toolsPermissionApplied=False, runtimeObserved=False)
open(p, "w", encoding="utf-8").write(dump(rows))
# 2) move projects
PM = {"P601": "P601_261023_의존성_기록", "P602": "P602_261120_릴리스_문서"}
for k, v in PM.items(): shutil.move(F + "/projects/" + k, F + "/projects/" + v)
# 3) backlog folder
BR = "backlogs/BL261112_02_허브_진행_메모"
B = F + "/" + BR
os.makedirs(B)
shutil.move(F + "/decisions", B + "/decisions")
shutil.move(F + "/review-records", B + "/review-records")
for root, _, fs in os.walk(F):
    for f in fs:
        if f.endswith((".md", ".json", ".jsonl", ".patch")):
            q = os.path.join(root, f)
            t = open(q, encoding="utf-8").read(); o = t
            for k, v in PM.items(): t = t.replace("projects/" + k + "/", "projects/" + v + "/")
            t = t.replace("](decisions/", "](" + BR + "/decisions/")
            if t != o: open(q, "w", encoding="utf-8").write(t)
# 4) backlog REVIEW-LOG
def rec(n): return json.load(open(B + "/review-records/R%d.json" % n, encoding="utf-8"))
def rv(n, verdict, re_of):
    r = rec(n)
    return dict(type="review", at=r["at"], kind=r["kind"], target="BACKLOG.md", hash=sha("synthetic-backlog-snapshot-R%d" % n),
                verdict=verdict, verifier=r["verifier"], provider="anthropic", rereview_of=re_of,
                record="review-records/R%d.json" % n, result_hash=sha(r["result"]))
L = [rv(1, "fail", None), rv(2, "fail", "review-records/R1.json"), rv(3, "fail", "review-records/R2.json"),
     dict(type="diagnosis", at="2026-11-16T09:00:00.000Z", target="BACKLOG.md", layer="planning",
          summary="R1~R3 지적이 모두 hub-notes의 결정 열·owner 열 작성이다", file="decisions/BL261112_02-diagnosis.md"),

          summary="hub-notes 출력 범위를 정의서와 맞춘다", file="decisions/BL261112_02-rule-change.md", by="pm"),
     rv(5, "pass", None)]
open(B + "/REVIEW-LOG.jsonl", "w", encoding="utf-8").write(dump(L))
